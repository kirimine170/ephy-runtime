cmake_minimum_required(VERSION 3.20)
project(ephy_whisper_worker LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT EXISTS "${WHISPER_SOURCE_DIR}/include/whisper.h")
  message(FATAL_ERROR "Set WHISPER_SOURCE_DIR to the pinned whisper.cpp checkout")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_IS_DEV OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
add_subdirectory("${WHISPER_SOURCE_DIR}" whisper EXCLUDE_FROM_ALL)
find_package(Threads REQUIRED)
add_executable(ephy-whisper worker.cpp)
target_include_directories(ephy-whisper PRIVATE "${WHISPER_SOURCE_DIR}/examples")
target_link_libraries(ephy-whisper PRIVATE whisper Threads::Threads)
target_compile_options(ephy-whisper PRIVATE -Wall -Wextra -Werror)
add_executable(ephy-resampler-test resampler_test.cpp)
target_compile_options(ephy-resampler-test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
enable_testing()
add_test(NAME resampler COMMAND ephy-resampler-test)
